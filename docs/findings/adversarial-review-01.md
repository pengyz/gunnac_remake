## 6. 复现命令

```bash
python3 - <<'PY'
import json
from collections import Counter
rows=[json.loads(l) for l in open('analysis/phase/gameplay-stage1-text.jsonl') if l.strip()]
print(Counter(e['addr'] for e in rows))                      # 2006:19512 2007:746960 2000:9496
bursts=[];pend=None;cur=None
for e in rows:
    if e['addr']=='2006':
        if pend is None: pend=e['val']
        else: cur={'a0':((pend&0x3F)<<8)|e['val'],'w':[]};bursts.append(cur);pend=None
    elif e['addr']=='2007' and cur is not None: cur['w'].append(e['val'])
print(len(bursts), Counter(len(b['w']) for b in bursts).most_common(8))   # 9756 / 77:9478 …
seen={};same=diff=0
for b in bursts:
    for i,v in enumerate(b['w']):
        a=(b['a0']+i)&0x3FFF
        if a in seen: same, diff = same+(seen[a]==v), diff+(seen[a]!=v)
        seen[a]=v
print(len(seen), same, diff)                                  # 2893 736655 7412
PY
```
